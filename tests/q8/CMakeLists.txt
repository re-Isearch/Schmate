cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(SchmatePassRegressions LANGUAGES CXX)
    enable_testing()
endif()

find_package(Threads REQUIRED)
if(TARGET sbert_core)
    set(pass_regression_core_target sbert_core)
else()
    # Exercise the production search and persistence code without a model or
    # inference backend, using the same focused sources as run_energy_tests.sh.
    add_library(pass_regression_core STATIC
        ../../src/unified_hnsw.cpp ../../src/HnswConfig.cpp
        ../../src/Logger.cpp ../../src/FileLock.cpp)
    target_compile_features(pass_regression_core PUBLIC cxx_std_20)
    target_include_directories(pass_regression_core PUBLIC ../../include)
    target_compile_options(pass_regression_core PRIVATE -ffunction-sections -fdata-sections)
    target_link_libraries(pass_regression_core PUBLIC Threads::Threads)
    if(APPLE)
        target_link_options(pass_regression_core INTERFACE -Wl,-dead_strip)
    else()
        target_link_options(pass_regression_core INTERFACE -Wl,--gc-sections)
    endif()
    set(pass_regression_core_target pass_regression_core)
endif()
foreach(test_target test_q8_passthrough test_float_passthrough)
    add_executable(${test_target} ../${test_target}.cpp)
    target_compile_features(${test_target} PRIVATE cxx_std_20)
    target_include_directories(${test_target} PRIVATE ../../include)
    target_link_libraries(${test_target} PRIVATE ${pass_regression_core_target} Threads::Threads)
endforeach()
add_test(NAME q8_passthrough COMMAND test_q8_passthrough)
add_test(NAME float_passthrough COMMAND test_float_passthrough)
